1//! Page metadata derived from chapter content: descriptions and canonical URLs.
2
3use anyhow::Result;
4use markdown::mdast::Node;
5use markdown::to_mdast;
6
7use crate::config::BookConfig;
8
9/// Longest description worth emitting; search engines truncate around here.
10const MAX_DESCRIPTION: usize = 200;
11
12/// Plain text of the first paragraph, for `<meta name="description">`.
13///
14/// Taken from the parsed AST rather than by scanning raw markdown, so emphasis,
15/// links and inline code flatten to their text and non-ASCII survives. An
16/// earlier implementation mapped every non-ASCII character to a space, which
17/// turned "café" into "caf ".
18///
19/// # Errors
20///
21/// Returns an error if the markdown cannot be parsed.
22pub fn first_paragraph_text(content: &str, config: &BookConfig) -> Result<Option<String>> {
23    fn collect_text(node: &Node, out: &mut String) {
24        match node {
25            Node::Text(t) => out.push_str(&t.value),
26            Node::InlineCode(c) => out.push_str(&c.value),
27            _ => {
28                if let Some(children) = node.children() {
29                    for child in children {
30                        collect_text(child, out);
31                    }
32                }
33            }
34        }
35    }
36
37    fn find_paragraph(node: &Node) -> Option<&Node> {
38        if matches!(node, Node::Paragraph(_)) {
39            return Some(node);
40        }
41        node.children()?.iter().find_map(find_paragraph)
42    }
43
44    let options = crate::render::markdown::options_for(config);
45    let ast = to_mdast(content, &options.parse)
46        .map_err(|e| anyhow::anyhow!("Markdown parsing error: {e:?}"))?;
47
48    let Some(paragraph) = find_paragraph(&ast) else {
49        return Ok(None);
50    };
51
52    let mut text = String::new();
53    collect_text(paragraph, &mut text);
54    let text = text.split_whitespace().collect::<Vec<_>>().join(" ");
55
56    if text.is_empty() {
57        return Ok(None);
58    }
59    Ok(Some(truncate_on_word(&text, MAX_DESCRIPTION)))
60}
61
62fn truncate_on_word(text: &str, max: usize) -> String {
63    if text.chars().count() <= max {
64        return text.to_string();
65    }
66    let truncated: String = text.chars().take(max).collect();
67    match truncated.rsplit_once(' ') {
68        Some((head, _)) if !head.is_empty() => format!("{head}…"),
69        _ => format!("{truncated}…"),
70    }
71}
72
73/// Description for a page: its first paragraph, else the book description,
74/// else the book title.
75pub fn page_description(content: &str, config: &BookConfig) -> String {
76    first_paragraph_text(content, config)
77        .ok()
78        .flatten()
79        .or_else(|| config.book.description.clone())
80        .unwrap_or_else(|| config.book.title.clone())
81}

83/// Absolute canonical URL for a page, or `None` when no site URL is configured.
84///
85/// Never returns a relative URL: a relative `<link rel="canonical">` is worse
86/// than none, since it resolves against whatever path served the page.
87#[must_use]
88pub fn canonical_url(site_prefix: &str, page_path: &str) -> Option<String> {
89    if site_prefix.is_empty() {
90        return None;
91    }
92    Some(format!(
93        "{site_prefix}{}",
94        page_path.trim_start_matches('/')
95    ))
96}

98#[cfg(test)]
99mod tests {
100    use super::*;
101
102    fn config() -> BookConfig {
103        BookConfig::default()
104    }
105
106    #[test]
107    fn test_description_from_first_paragraph() {
108        let md = "# Title\n\nThe **first** paragraph with `code` and a [link](x.md).\n\nSecond.";
109        let desc = page_description(md, &config());
110        assert_eq!(desc, "The first paragraph with code and a link.");
111    }
112
113    #[test]
114    fn test_description_preserves_non_ascii() {
115        let md = "# Café\n\nUn café au lait, s'il vous plaît. Обзор. 日本語.";
116        let desc = page_description(md, &config());
117        assert!(desc.contains("café"), "{desc}");
118        assert!(desc.contains("Обзор"), "{desc}");
119        assert!(desc.contains("日本語"), "{desc}");
120    }
121
122    #[test]
123    fn test_description_falls_back_to_book_then_title() {
124        let mut c = config();
125        c.book.description = Some("Book level".into());
126        c.book.title = "The Title".into();
127
128        assert_eq!(page_description("# Only a heading\n", &c), "Book level");
129
130        c.book.description = None;
131        assert_eq!(page_description("# Only a heading\n", &c), "The Title");
132    }
133
134    #[test]
135    fn test_description_is_truncated_on_a_word_boundary() {
136        let long = "word ".repeat(100);
137        let md = format!("# T\n\n{long}");
138        let desc = page_description(&md, &config());
139        assert!(
140            desc.chars().count() <= MAX_DESCRIPTION + 1,
141            "{}",
142            desc.len()
143        );
144        assert!(desc.ends_with('…'));
145        assert!(!desc.contains("wor…"), "should cut between words: {desc}");
146    }
147
148    #[test]
149    fn test_canonical_absent_without_site_url() {
150        assert_eq!(canonical_url("", "a.html"), None);
151    }
152
153    #[test]
154    fn test_canonical_joins_once() {
155        assert_eq!(
156            canonical_url("https://example.com/docs/", "/a/b.html"),
157            Some("https://example.com/docs/a/b.html".into())
158        );
159    }
