322/// A configuration key md-book accepts but does not act on.
323#[derive(Debug, Clone, PartialEq, Eq)]
324pub struct UnsupportedKey {
325    /// Dotted path as it appears in `book.toml`.
326    pub path: String,
327    /// Why it has no effect.
328    pub reason: &'static str,
329}
330
331/// Keys that parse but change nothing, with the reason to report.
332///
333/// A prefix entry covers its whole table: `output.html.playground` also reports
334/// `output.html.playground.editable`.
335const UNSUPPORTED: &[(&str, &str)] = &[
336    (
337        "output.html.mathjax-support",
338        "MathJax rendering is not implemented",
339    ),
340    (
341        "output.html.playground",
342        "the Rust Playground runtime is out of scope for md-book",
343    ),
344    (
345        "output.html.search.use-boolean-and",
346        "md-book searches with Pagefind, which has no equivalent setting",
347    ),
348    (
349        "output.html.search.boost-title",
350        "md-book searches with Pagefind, which has no equivalent setting",
351    ),
352    (
353        "output.html.search.boost-hierarchy",
354        "md-book searches with Pagefind, which has no equivalent setting",
355    ),
356    (
357        "output.html.search.boost-paragraph",
358        "md-book searches with Pagefind, which has no equivalent setting",
359    ),
360    (
361        "output.html.search.expand",
362        "md-book searches with Pagefind, which has no equivalent setting",
363    ),
364    (
365        "output.html.search.teaser-word-count",
366        "md-book searches with Pagefind, which has no equivalent setting",
367    ),
368];

370/// Find unsupported keys the author actually set in a config document.
371///
372/// Works on the parsed document rather than the loaded `BookConfig` because a
373/// deserialised default is indistinguishable from a value the author typed --
374/// warning on defaults would fire for every book.
375#[must_use]
376pub fn unsupported_keys_in(doc: &serde_json::Value) -> Vec<UnsupportedKey> {
377    let mut found = Vec::new();
378
379    for (path, reason) in UNSUPPORTED {
380        let mut cursor = doc;
381        let mut present = true;
382        for segment in path.split('.') {
383            match cursor.get(segment) {
384                Some(next) => cursor = next,
385                None => {
386                    present = false;
387                    break;
388                }
389            }
390        }
391        if present && !cursor.is_null() {
392            found.push(UnsupportedKey {
393                path: (*path).to_string(),
394                reason,
395            });
396        }
397    }
398
399    found
400}

402/// Read a config file and report any keys that have no effect.
403///
404/// Failure to read or parse is silent: the layering itself will surface a
405/// genuine problem, and this is only advisory.
406fn warn_unsupported_keys(path: &Path) {
407    let Ok(text) = std::fs::read_to_string(path) else {
408        return;
409    };
410
411    let doc: Option<serde_json::Value> = match path.extension().and_then(|e| e.to_str()) {
412        Some(ext) if ext.eq_ignore_ascii_case("toml") => toml::from_str(&text).ok(),
413        Some(ext) if ext.eq_ignore_ascii_case("json") => serde_json::from_str(&text).ok(),
414        _ => None,
415    };
416    let Some(doc) = doc else { return };
417
418    for key in unsupported_keys_in(&doc) {
419        eprintln!(
420            "warning: {}: '{}' is ignored -- {}",
421            path.display(),
422            key.path,
423            key.reason
424        );
425    }
426}

428/// Load configuration from file or use defaults
429///
430/// # Errors
431///

433pub fn load_config(config_path: Option<&str>) -> anyhow::Result<BookConfig> {
434    load_config_from(None, config_path)
435}

437/// Load configuration for a book directory.
438///
439/// Layers, lowest precedence first: `MDBOOK_` environment variables, the
440/// book directory's `book.toml` (or the current directory's when `book_dir` is
441/// `None`), then an explicit config file. Unset scalars are then filled from the
442/// documented defaults, and keys that have no effect are reported.
443///
444/// # Errors
445///
446/// Returns an error if a layer cannot be read or parsed, or if `config_path` has
447/// an extension other than `.toml` or `.json`.
448pub fn load_config_from(
449    book_dir: Option<&Path>,
450    config_path: Option<&str>,
451) -> anyhow::Result<BookConfig> {
452    let mut layers = vec![Layer::Env(Some("MDBOOK_".to_string()))];
453
454    let book_toml = book_dir
455        .map(|dir| dir.join("book.toml"))
456        .unwrap_or_else(|| PathBuf::from("book.toml"));
457    if book_toml.exists() {
458        warn_unsupported_keys(&book_toml);
459        layers.push(Layer::Toml(book_toml));
460    }
461
462    if let Some(path) = config_path {
463        let path = Path::new(path);
464        if path.exists() {
465            let ext = path
466                .extension()
467                .and_then(|e| e.to_str())
468                .unwrap_or_default();
469            if ext.eq_ignore_ascii_case("toml") {
470                warn_unsupported_keys(path);
471                layers.push(Layer::Toml(path.to_path_buf()));
472            } else if ext.eq_ignore_ascii_case("json") {
473                warn_unsupported_keys(path);
474                layers.push(Layer::Json(path.to_path_buf()));
475            } else {
476                anyhow::bail!("Unsupported config file type: {}", path.display());
477            }
478        }
479    }
480
481    let mut config = BookConfig::with_layers(&layers)?;
482    fill_unset_with_defaults(&mut config);
483    Ok(config)
484}
485
486/// Replace unset scalar fields with their documented defaults.
487///
488/// `#[serde(default = "...")]` does not survive two hops here: twelf's layering
489/// yields empty strings for absent keys, and `#[serde(default)]` on a container
490/// field constructs it with `Default::default()`, bypassing the per-field
491/// defaults inside it. Without this, a book with no `book.toml` renders
492/// `<html lang="">`, an empty `<title>` and a broken logo `src=""`.
493fn fill_unset_with_defaults(config: &mut BookConfig) {
494    let d = BookConfig::default();
495
496    if config.book.title.is_empty() {
497        config.book.title = d.book.title;
498    }
499    if config.book.language.is_empty() {
500        config.book.language = d.book.language;
501    }
502    if config.book.logo.is_empty() {
503        config.book.logo = d.book.logo;
504    }
505    if config.rust.edition.is_empty() {
506        config.rust.edition = d.rust.edition;
507    }
508    if config.paths.templates.is_empty() {
509        config.paths.templates = d.paths.templates;
510    }
511
512    // Zero is never a meaningful value for these, so it means "absent".
513    let search = &mut config.output.html.search;
514    let ds = d.output.html.search;
515    if search.limit_results == 0 {
516        search.limit_results = ds.limit_results;
517    }
518    if search.heading_split_level == 0 {
519        search.heading_split_level = ds.heading_split_level;
520    }
521}
