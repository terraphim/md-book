1//! Directory-walk book construction (fallback when no SUMMARY.md exists).
2
3use std::collections::BTreeMap;
4use std::fs;
5use std::path::Path;
6
7use anyhow::Result;
8
9use super::summary::source_to_output;
10use super::{Book, BookItem, Chapter};
11use crate::render::html::extract_title;
12
13/// Today's behaviour: walk `src_dir`, sort by path, group by parent directory.
14/// Used when no `SUMMARY.md` exists.
15pub fn book_from_directory(src_dir: &Path) -> Result<Book> {
16    let mut entries: Vec<_> = walkdir::WalkDir::new(src_dir)
17        .into_iter()
18        .filter_map(Result::ok)
19        .filter(|e| e.path().extension().is_some_and(|ext| ext == "md"))
20        .collect();
21
22    entries.sort_by_key(|e| e.path().to_path_buf());
23
24    // Group by parent directory for legacy sections, but produce a flat
25    // chapter list in path order (matches previous prev/next behaviour).
26    let mut root_chapters: Vec<BookItem> = Vec::new();
27    let mut by_parent: BTreeMap<String, Vec<BookItem>> = BTreeMap::new();
28
29    for entry in &entries {
30        let rel = entry.path().strip_prefix(src_dir)?;
31        let parent = rel
32            .parent()
33            .and_then(|p| p.to_str())
34            .unwrap_or("")
35            .to_string();
36
37        let content = fs::read_to_string(entry.path())?;
38        let name = extract_title(&content).unwrap_or_else(|| {
39            entry.path().file_stem().map_or_else(
40                || "Untitled".to_string(),
41                |s| s.to_string_lossy().into_owned(),
42            )
43        });
44
45        let chapter = Chapter {
46            name,
47            source_path: Some(rel.to_path_buf()),
48            output_path: Some(source_to_output(rel)),
49            number: None,
50            sub_items: Vec::new(),
51            fragment: None,
52            is_external: false,
53            external_url: None,
54        };
55
56        let item = BookItem::Chapter(chapter);
57        if parent.is_empty() {
58            root_chapters.push(item);
59        } else {
60            by_parent.entry(parent).or_default().push(item);
61        }
62    }
63
64    // Preserve previous ordering: root pages first (as "Guide" section pages),
65    // then each parent directory group. Represent as flat top-level chapters so
66    // prev/next follows path sort within each group, and overall matches old
67    // all_pages order (path-sorted).
68    //
69    // Old code path-sorted *all* entries globally then built sections. Prev/next
70    // used that global order. So items must be path-sorted flat list.
71    let mut all_items: Vec<BookItem> = Vec::new();
72    for entry in &entries {
73        let rel = entry.path().strip_prefix(src_dir)?;
74        let content = fs::read_to_string(entry.path())?;
75        let name = extract_title(&content).unwrap_or_else(|| {
76            entry.path().file_stem().map_or_else(
77                || "Untitled".to_string(),
78                |s| s.to_string_lossy().into_owned(),
79            )
80        });
81        all_items.push(BookItem::Chapter(Chapter {
82            name,
83            source_path: Some(rel.to_path_buf()),
84            output_path: Some(source_to_output(rel)),
85            number: None,
86            sub_items: Vec::new(),
87            fragment: None,
88            is_external: false,
89            external_url: None,
90        }));
91    }
92
93    let _ = (root_chapters, by_parent); // reserved for legacy section titles if needed
94
95    Ok(Book {
96        items: all_items,
97        from_summary: false,
98    })
99}
