22/// What a build did, beyond writing the output tree.
23#[derive(Debug, Default, Clone)]
24pub struct BuildReport {
25    /// Stub chapters written into the source tree by `build.create-missing`.
26    /// The watcher records these so it can ignore the events they cause.
27    pub created: Vec<std::path::PathBuf>,
28}

30/// Run the synchronous pipeline stages.
31pub fn run_sync(args: &Args, config: &BookConfig, watch_enabled: bool) -> Result<BuildReport> {
32    let tera = init_tera(config)?;
33    fs::create_dir_all(&args.output)?;
34    copy_static_assets(&args.output, &config.paths.templates, config)?;
35    let asset_version = theme_asset_version(&args.output)?;
36
37    let src_dir = Path::new(&args.input);
38    let create_missing = config.build.create_missing;
39    let (book, created) = load_book(src_dir, create_missing)?;
40
41    let chapters: Vec<&crate::book::Chapter> = book.iter_chapters().collect();
42    println!("Total pages: {}", chapters.len());
43
44    // Warn about orphan .md files when using SUMMARY
45    if book.from_summary {
46        warn_orphan_markdown(src_dir, &book, config)?;
47        // Copy non-markdown assets through
48        copy_non_markdown_assets(src_dir, Path::new(&args.output))?;
49    }
50
51    let additional = copy_additional_assets(config, src_dir, Path::new(&args.output))?;
52    // Canonical URLs need an absolute base; without one we emit none at all.
53    let site_prefix = site_url_prefix(config);
54    // Pagefind is indexed after this render. `core::build_impl` renders a
55    // second time after a successful index so a clean first build gets UI too.
56    let search_enabled = search_index_available(&args.output);
57
58    let sections = if book.from_summary {
59        book.to_legacy_sections()
60    } else {
61        legacy_directory_sections(&book)
62    };
63
64    let current_year = Zoned::now().year().to_string();
65    let no_section_label = config.output.html.no_section_label;
66    let fold = Some((
67        config.output.html.fold.enable,
68        config.output.html.fold.level,
69    ));
70
71    #[cfg(feature = "syntax-highlighting")]
72    let ss = {
73        use syntect::parsing::SyntaxSet;
74        SyntaxSet::load_defaults_newlines()
75    };
76
77    // Unconditional: the page template links css/syntax.css whether or not the
78    // feature is on, so the file has to exist either way.
79    write_syntax_css(&args.output, config)?;
80
81    let preprocess_ctx = PreprocessCtx;
82
83    // Prev/next over page-producing chapters only
84    for (idx, chapter) in chapters.iter().enumerate() {
85        let source_rel = chapter
86            .source_path
87            .as_ref()
88            .expect("iter_chapters only yields chapters with source");
89        let source_abs = src_dir.join(source_rel);
90        let output_rel = chapter
91            .output_path
92            .as_ref()
93            .expect("source chapter has output path");
94        let html_path = Path::new(&args.output).join(output_rel);
95
96        if let Some(parent) = html_path.parent() {
97            fs::create_dir_all(parent)?;
98        }
99
100        let markdown_content = fs::read_to_string(&source_abs)
101            .with_context(|| format!("Failed to read {}", source_abs.display()))?;
102        let preprocessed = preprocess(&markdown_content, &preprocess_ctx)?;
103
104        #[cfg(feature = "syntax-highlighting")]
105        let rendered = render_markdown(&preprocessed, config, Some(&ss))?;
106        #[cfg(not(feature = "syntax-highlighting"))]
107        let rendered = render_markdown(&preprocessed, config, None)?;
108        let html_content = crate::render::inject_heading_ids(&rendered.html);
109        let page_has_mermaid = rendered.has_mermaid;
110
111        // Prev/next links are relative to this page, like every other URL.
112        let page_root = crate::render::html::path_to_root(output_rel);
113        let previous = if idx > 0 {
114            Some(chapter_to_pageinfo(chapters[idx - 1], &page_root))
115        } else {
116            None
117        };
118        let next = if idx + 1 < chapters.len() {
119            Some(chapter_to_pageinfo(chapters[idx + 1], &page_root))
120        } else {
121            None
122        };
123
124        // SUMMARY link text wins as title
125        let title = flatten_title(&chapter.name);
126
127        let nav = book.to_nav(output_rel, no_section_label, fold);
128
129        render_page(
130            &tera,
131            &current_year,
132            config,
133            &additional,
134            watch_enabled,
135            &asset_version,
136            crate::render::PageRender {
137                html_path: &html_path.to_string_lossy(),
138                title,
139                description: crate::render::page_description(&markdown_content, config),
140                canonical_url: crate::render::canonical_url(
141                    &site_prefix,
142                    &crate::render::to_url_path(output_rel),
143                ),
144                content: html_content,
145                sections: &sections,
146                previous,
147                next,
148                current_path: &crate::render::to_url_path(output_rel),
149                chapters: Some(&nav),
150                has_mermaid: page_has_mermaid,
151                search_enabled,
152            },
153        )?;
154    }
155
156    // Index page: prefer a chapter whose output is index.html
157    let index_chapter = chapters.iter().find(|c| {
158        c.output_path
159            .as_ref()
160            .map(|p| p.as_os_str() == "index.html")
161            .unwrap_or(false)
162    });
163
164    let (index_page_info, index_content, index_has_mermaid, index_description) =
165        if let Some(ch) = index_chapter {
166            let source_abs = src_dir.join(ch.source_path.as_ref().unwrap());
167            let markdown_content = fs::read_to_string(&source_abs)?;
168            let preprocessed = preprocess(&markdown_content, &preprocess_ctx)?;
169            #[cfg(feature = "syntax-highlighting")]
170            let rendered = render_markdown(&preprocessed, config, Some(&ss))?;
171            #[cfg(not(feature = "syntax-highlighting"))]
172            let rendered = render_markdown(&preprocessed, config, None)?;
173            let html = crate::render::inject_heading_ids(&rendered.html);
174            // index.html sits at the build root, so links from it need no prefix.
175            (
176                Some(chapter_to_pageinfo(ch, "")),
177                Some(html),
178                rendered.has_mermaid,
179                crate::render::page_description(&markdown_content, config),
180            )
181        } else {
182            // Fallback: look for index.md even if not in book (directory edge case)
183            let index_path = src_dir.join("index.md");
184            if index_path.exists() {
185                let markdown_content = fs::read_to_string(&index_path)?;
186                let preprocessed = preprocess(&markdown_content, &preprocess_ctx)?;
187                #[cfg(feature = "syntax-highlighting")]
188                let rendered = render_markdown(&preprocessed, config, Some(&ss))?;
189                #[cfg(not(feature = "syntax-highlighting"))]
190                let rendered = render_markdown(&preprocessed, config, None)?;
191                let html = crate::render::inject_heading_ids(&rendered.html);
192                let info = PageInfo {
193                    title: "Documentation".into(),
194                    path: "/index.html".into(),
195                };
196                (
197                    Some(info),
198                    Some(html),
199                    rendered.has_mermaid,
200                    crate::render::page_description(&markdown_content, config),
201                )
202            } else {
203                (
204                    None,
205                    None,
206                    false,
207                    config
208                        .book
209                        .description
210                        .clone()
211                        .unwrap_or_else(|| config.book.title.clone()),
212                )
213            }
214        };
215
216    render_index(
217        &tera,
218        &args.output,
219        index_page_info.as_ref(),
220        index_content,
221        &sections,
222        &current_year,
223        config,
224        Some(&book.to_nav(Path::new("index.html"), no_section_label, fold)),
225        index_has_mermaid,
226        &additional,
227        index_description,
228        crate::render::canonical_url(&site_prefix, "index.html"),
229        search_enabled,
230        &asset_version,
231    )?;
232
233    // 404 page
234    {
235        // A server returns 404.html for *any* unmatched path, and the browser
236        // resolves relative URLs against the request path -- so a relative
237        // asset href breaks for every nested URL, which is exactly when a 404
238        // is served. When site-url is known, emit absolute paths instead.
239        let root_prefix = site_url_prefix(config);
240
241        // Optional custom body from `output.html.input-404`.
242        let custom_404 = match config.output.html.input_404.as_deref() {
243            None => None,
244            Some(name) => {
245                let path = src_dir.join(name);
246                match render_custom_404(&path, config, &preprocess_ctx, {
247                    #[cfg(feature = "syntax-highlighting")]
248                    {
249                        Some(&ss)
250                    }
251                    #[cfg(not(feature = "syntax-highlighting"))]
252                    {
253                        None
254                    }
255                }) {
256                    Ok(html) => Some(html),
257                    Err(e) => {
258                        eprintln!(
259                            "warning: input-404 '{}' could not be rendered, \
260                             using the built-in 404 body: {e}",
261                            path.display()
262                        );
263                        None
264                    }
265                }
266            }
267        };
268
269        let mut ctx = tera::Context::new();
270        ctx.insert("config", &config);
271        ctx.insert("path_to_root", &root_prefix);
272        ctx.insert("custom_404", &custom_404);
273        ctx.insert("additional_css", &additional.css);
274        ctx.insert("additional_js", &additional.js);
275        ctx.insert("asset_version", &asset_version);
276        ctx.insert("year", &current_year);
277        ctx.insert("default_theme", &config.output.html.default_theme_name());
278        ctx.insert(
279            "preferred_dark_theme",
280            &config.output.html.preferred_dark_theme_name(),
281        );
282
283        match tera.render("404.html", &ctx) {
284            Ok(html) => {
285                if let Err(e) = fs::write(format!("{}/404.html", args.output), html) {
286                    eprintln!("warning: failed to write 404.html: {e}");
287                }
288            }
289            Err(e) => eprintln!("warning: failed to render 404 template: {e}"),
290        }
291    }
292
293    // Redirect stubs (path-contained + HTML-escaped)
294    let out_root = Path::new(&args.output);
295    for (from, to) in &config.output.html.redirect {
296        let rel = from.trim_start_matches('/');
297        // Reject absolute / parent-escaping redirect sources
298        let rel_path = Path::new(rel);
299        if rel_path.is_absolute()
300            || rel_path
301                .components()
302                .any(|c| matches!(c, std::path::Component::ParentDir))
303        {
304            eprintln!("warning: ignoring unsafe redirect source path: {from}");
305            continue;
306        }
307        let dest = out_root.join(rel_path);
308        // Containment check after join
309        let dest_canon_parent = dest.parent().and_then(|p| {
310            let _ = fs::create_dir_all(p);
311            p.canonicalize().ok()
312        });
313        let out_canon = out_root.canonicalize().ok();
314        if let (Some(parent), Some(root)) = (dest_canon_parent, out_canon) {
315            if !parent.starts_with(&root) {
316                eprintln!("warning: redirect escapes build dir, skipped: {from}");
317                continue;
318            }
319        }
320        // Only allow relative, root-relative, or http(s) targets
321        let to_safe = to.trim();
322        let to_ok = to_safe.starts_with('/')
323            || to_safe.starts_with("./")
324            || to_safe.starts_with("../")
325            || to_safe.starts_with("http://")
326            || to_safe.starts_with("https://")
327            || (!to_safe.contains(':') && !to_safe.is_empty());
328        if !to_ok || to_safe.contains('\n') || to_safe.contains('"') || to_safe.contains('<') {
329            eprintln!("warning: ignoring unsafe redirect target: {to}");
330            continue;
331        }
332        let escaped = html_escape::encode_double_quoted_attribute(to_safe);
333        let body = format!(
334            r#"<!DOCTYPE html><html><head><meta http-equiv="refresh" content="0; url={escaped}"><link rel="canonical" href="{escaped}"></head><body><a href="{escaped}">Redirect</a></body></html>"#
335        );
336        if let Err(e) = fs::write(&dest, body) {
337            eprintln!("warning: failed to write redirect {from}: {e}");
338        }
339    }
340
341    // Print page (all chapters concatenated)
342    if config.output.html.print.enable {
343        #[derive(serde::Serialize)]
344        struct PrintChapter {
345            title: String,
346            content: String,
347            anchor: String,
348        }
349        let mut print_chapters = Vec::new();
350        let mut print_has_mermaid = false;
351        // All chapters land in one document, so they share one id namespace.
352        let mut print_ids: std::collections::HashMap<String, usize> =
353            std::collections::HashMap::new();
354        for ch in &chapters {
355            let source_abs = src_dir.join(ch.source_path.as_ref().unwrap());
356            match fs::read_to_string(&source_abs) {
357                Err(e) => eprintln!("warning: print page omits '{}': {e}", source_abs.display()),
358                Ok(md) => match preprocess(&md, &preprocess_ctx) {
359                    Err(e) => {
360                        eprintln!("warning: print page omits '{}': {e}", source_abs.display())
361                    }
362                    Ok(pre) => {
363                        #[cfg(feature = "syntax-highlighting")]
364                        let rendered = render_markdown(&pre, config, Some(&ss))?;
365                        #[cfg(not(feature = "syntax-highlighting"))]
366                        let rendered = render_markdown(&pre, config, None)?;
367                        let html =
368                            crate::render::inject_heading_ids_with(&mut print_ids, &rendered.html);
369                        print_has_mermaid |= rendered.has_mermaid;
370                        print_chapters.push(PrintChapter {
371                            title: flatten_title(&ch.name),
372                            content: html,
373                            anchor: ch
374                                .output_path
375                                .as_ref()
376                                .map(|p| crate::render::to_url_path(p))
377                                .unwrap_or_default(),
378                        });
379                    }
380                },
381            }
382        }
383        let mut ctx = tera::Context::new();
384        ctx.insert("config", &config);
385        ctx.insert("path_to_root", &"");
386        ctx.insert("print_chapters", &print_chapters);
387        ctx.insert("page_break", &config.output.html.print.page_break);
388        ctx.insert("has_mermaid", &print_has_mermaid);
389        ctx.insert("additional_css", &additional.css);
390        ctx.insert("additional_js", &additional.js);
391        ctx.insert("asset_version", &asset_version);
392        ctx.insert("default_theme", &config.output.html.default_theme_name());
393        ctx.insert(
394            "preferred_dark_theme",
395            &config.output.html.preferred_dark_theme_name(),
396        );
397
398        match tera.render("print.html", &ctx) {
399            Ok(html) => {
400                if let Err(e) = fs::write(format!("{}/print.html", args.output), html) {
401                    eprintln!("warning: failed to write print.html: {e}");
402                }
403            }
404            Err(e) => eprintln!("warning: failed to render print.html: {e}"),
405        }
406    }
407
408    #[cfg(not(all(feature = "search", feature = "tokio")))]
409    {
410        println!("Skipping search indexing (search or tokio feature not enabled)");
411    }
412
413    Ok(BuildReport { created })
414}

416/// Render the configured `input-404` markdown into an HTML fragment.
417fn render_custom_404(
418    path: &Path,
419    config: &BookConfig,
420    preprocess_ctx: &PreprocessCtx,
421    #[cfg(feature = "syntax-highlighting")] syntax_set: Option<&syntect::parsing::SyntaxSet>,
422    #[cfg(not(feature = "syntax-highlighting"))] syntax_set: Option<&()>,
423) -> Result<String> {
424    let markdown =
425        fs::read_to_string(path).with_context(|| format!("Failed to read {}", path.display()))?;
426    let preprocessed = preprocess(&markdown, preprocess_ctx)?;
427    let rendered = render_markdown(&preprocessed, config, syntax_set)?;
428    Ok(crate::render::inject_heading_ids(&rendered.html))
429}

431/// Whether a Pagefind index exists in the output from a previous build.
432///
433/// Indexing runs after HTML generation, so on a first build with search enabled
434/// this is false and the UI appears from the second build onwards -- the same
435/// trade-off mdBook makes by shipping its index alongside.
436fn search_index_available(output_dir: &str) -> bool {
437    Path::new(output_dir).join("pagefind/pagefind.js").exists()
438}

440/// Remove a stale or partially written Pagefind index after indexing failure.
441pub fn discard_search_index(output_dir: &str) -> Result<()> {
442    let path = Path::new(output_dir).join("pagefind");
443    if path.exists() {
444        fs::remove_dir_all(path).context("Failed to discard stale Pagefind index")?;
445    }
446    Ok(())

449/// Copy `additional-css` / `additional-js` into the build and return their
450/// build-root-relative paths, in the order the author listed them.
451///
452/// A missing file is reported rather than skipped in silence: the author asked
453/// for a stylesheet and would otherwise get an unstyled page with no clue why.
454fn copy_additional_assets(
455    config: &BookConfig,
456    src_dir: &Path,
457    out_dir: &Path,
458) -> Result<AdditionalAssets> {
459    fn copy_list(list: &[String], src_dir: &Path, out_dir: &Path, kind: &str) -> Vec<String> {
460        let mut urls = Vec::new();
461        for entry in list {
462            // Authors write paths relative to the book root; accept a path
463            // relative to src/ too, since that is where their files usually sit.
464            let candidates = [src_dir.join(entry), src_dir.join("..").join(entry)];
465            let Some(source) = candidates.iter().find(|p| p.is_file()) else {
466                eprintln!("warning: {kind} '{entry}' not found; skipping");
467                continue;
468            };
469
470            let file_name = match Path::new(entry).file_name() {
471                Some(name) => name,
472                None => continue,
473            };
474            let dest_rel = Path::new("additional").join(file_name);
475            let dest = out_dir.join(&dest_rel);
476            if let Some(parent) = dest.parent() {
477                if let Err(e) = fs::create_dir_all(parent) {
478                    eprintln!("warning: cannot create {}: {e}", parent.display());
479                    continue;
480                }
481            }
482            if let Err(e) = fs::copy(source, &dest) {
483                eprintln!("warning: cannot copy {kind} '{entry}': {e}");
484                continue;
485            }
486            urls.push(crate::render::to_url_path(&dest_rel));
487        }
488        urls
489    }
490
491    Ok(AdditionalAssets {
492        css: copy_list(
493            &config.output.html.additional_css,
494            src_dir,
495            out_dir,
496            "additional-css",
497        ),
498        js: copy_list(
499            &config.output.html.additional_js,
500            src_dir,
501            out_dir,
502            "additional-js",
503        ),
504    })
505}

514/// Absolute prefix for pages that may be served from an unknown path.
515///
516/// Returns `output.html.site-url` (or the deprecated `book.base_url`) with
517/// exactly one trailing slash, or an empty string when neither is configured --
518/// in which case relative paths are the best available answer.
519fn site_url_prefix(config: &BookConfig) -> String {
520    let configured = config
521        .output
522        .html
523        .site_url
524        .as_deref()
525        .or(config.book.base_url.as_deref());
526
527    match configured {
528        Some(url) if !url.is_empty() => {
529            let trimmed = url.trim_end_matches('/');
530            format!("{trimmed}/")
531        }
532        _ => String::new(),
533    }
534}

536fn chapter_to_pageinfo(ch: &crate::book::Chapter, root: &str) -> PageInfo {
537    // Marked `| safe` in the templates, so it must be safe by construction:
538    // output paths come from filenames, which may contain a quote.
539    PageInfo {
540        title: flatten_title(&ch.name),
541        path: crate::render::escape_url_attr(&format!(
542            "{root}{}",
543            ch.output_path
544                .as_ref()
545                .map(|p| crate::render::to_url_path(p))
546                .unwrap_or_default()
547        )),
548    }
549}

551/// Directory-mode sections: root → "Guide", other parents → dir name (old behaviour).
552fn legacy_directory_sections(book: &Book) -> Vec<Section> {
553    use std::collections::BTreeMap;
554    let mut section_map: BTreeMap<String, Vec<PageInfo>> = BTreeMap::new();
555    let mut root_pages = Vec::new();
556
557    for ch in book.iter_chapters() {
558        let out = match &ch.output_path {
559            Some(p) => p,
560            None => continue,
561        };
562        // Deprecated flat `sections` are built once per book, not per page, so
563        // their paths are stored relative to the build root and templates
564        // prefix them with `path_to_root`. Removed in 0.3.0.
565        let info = chapter_to_pageinfo(ch, "");
566        let parent = ch
567            .source_path
568            .as_ref()
569            .and_then(|p| p.parent())
570            .and_then(|p| p.to_str())
571            .unwrap_or("");
572        if parent.is_empty() {
573            root_pages.push(info);
574        } else {
575            section_map
576                .entry(parent.to_string())
577                .or_default()
578                .push(info);
579        }
580        let _ = out;
581    }
582
583    let mut sections = Vec::new();
584    if !root_pages.is_empty() {
585        sections.push(Section {
586            title: "Guide".to_string(),
587            pages: root_pages,
588        });
589    }
590    for (title, pages) in section_map {
591        sections.push(Section { title, pages });
592    }
593    sections
594}

596fn warn_orphan_markdown(src_dir: &Path, book: &Book, config: &BookConfig) -> Result<()> {
597    use std::collections::HashSet;
598    let mut listed: HashSet<PathBuf> = book
599        .iter_all_chapters()
600        .filter_map(|c| c.source_path.clone())
601        .collect();
602
603    // The 404 source is consumed by the 404 page, not by the summary, so it is
604    // not an orphan even though no chapter references it.
605    if let Some(input_404) = config.output.html.input_404.as_deref() {
606        listed.insert(PathBuf::from(input_404));
607    }
608
609    for entry in WalkDir::new(src_dir).into_iter().filter_map(Result::ok) {
610        let path = entry.path();
611        if path.extension().and_then(|e| e.to_str()) != Some("md") {
612            continue;
613        }
614        let rel = path.strip_prefix(src_dir)?;
615        if rel == Path::new("SUMMARY.md") {
616            continue;
617        }
618        if !listed.contains(rel) {
619            eprintln!(
620                "warning: markdown file not in SUMMARY.md (not published): {}",
621                rel.display()
622            );
623        }
624    }
625    Ok(())
626}

628fn copy_non_markdown_assets(src_dir: &Path, out_dir: &Path) -> Result<()> {
629    for entry in WalkDir::new(src_dir).into_iter().filter_map(Result::ok) {
630        if !entry.file_type().is_file() {
631            continue;
632        }
633        let path = entry.path();
634        if path.extension().and_then(|e| e.to_str()) == Some("md") {
635            continue;
636        }
637        let rel = path.strip_prefix(src_dir)?;
638        let dest = out_dir.join(rel);
639        if let Some(parent) = dest.parent() {
640            fs::create_dir_all(parent)?;
641        }
642        fs::copy(path, &dest)
643            .with_context(|| format!("Failed to copy asset {}", path.display()))?;
644    }
645    Ok(())
646}

648/// Run Pagefind indexing over the output directory (async path only).
649#[cfg(all(feature = "search", feature = "tokio"))]
650pub async fn index(output: &str) -> Result<bool> {
651    use crate::pagefind_service::PagefindBuilder;
652
653    match PagefindBuilder::new(PathBuf::from(output)).await {
654        Ok(pagefind) => match pagefind.build().await {
655            Ok(()) => return Ok(true),
656            Err(e) => eprintln!("Search indexing failed: {e}"),
657        },
658        Err(e) => {
659            eprintln!("Failed to create search builder: {e}");
660        }
661    }
662    Ok(false)
663}
